import random

# Initialize Q-table
Q = {}

# Game board
def create_board():
    return [' '] * 9

# Check winner
def winner(board, player):
    wins = [
        (0,1,2), (3,4,5), (6,7,8),
        (0,3,6), (1,4,7), (2,5,8),
        (0,4,8), (2,4,6)
    ]

    for a, b, c in wins:
        if board[a] == board[b] == board[c] == player:
            return True
    return False

# Get available actions
def actions(board):
    return [i for i in range(9) if board[i] == ' ']

# Choose action
def choose_action(state, epsilon):
    available = actions(list(state))

    if random.random() < epsilon:
        return random.choice(available)

    values = [Q.get((state, a), 0) for a in available]
    best = max(values)

    choices = [
        a for a in available
        if Q.get((state, a), 0) == best
    ]

    return random.choice(choices)

# Train the agent
def train():
    global Q

    for episode in range(50000):
        board = create_board()
        history = []

        while True:
            state = tuple(board)

            # Agent plays X
            action = choose_action(state, 0.2)
            board[action] = 'X'
            history.append((state, action))

            if winner(board, 'X'):
                reward = 1
                break

            if not actions(board):
                reward = 0
                break

            # Random opponent plays O
            opponent = random.choice(actions(board))
            board[opponent] = 'O'

            if winner(board, 'O'):
                reward = -1
                break

            if not actions(board):
                reward = 0
                break

        # Update Q-table
        for state, action in reversed(history):
            old = Q.get((state, action), 0)
            Q[(state, action)] = old + 0.1 * (reward - old)
            reward *= 0.9

    print("Training completed!")

# Display board
def display(board):
    print()
    for i in range(0, 9, 3):
        print(" | ".join(board[i:i+3]))
        if i < 6:
            print("---------")

# Play game
def play():
    board = create_board()

    print("\nYou are O. Computer is X.")

    while True:
        # Computer move
        state = tuple(board)
        move = choose_action(state, 0)

        board[move] = 'X'
        display(board)

        if winner(board, 'X'):
            print("Computer wins!")
            break

        if not actions(board):
            print("Match Draw!")
            break

        # Human move
        while True:
            try:
                move = int(input("Enter position (0-8): "))
                if move in actions(board):
                    break
                print("Invalid position. Try again.")
            except ValueError:
                print("Enter a number from 0 to 8.")

        board[move] = 'O'
        display(board)

        if winner(board, 'O'):
            print("You win!")
            break

        if not actions(board):
            print("Match Draw!")
            break

# Main program
train()
play()
